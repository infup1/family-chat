            const txt = document.getElementById('syncStatusText');


            try {

                mqttClient.on('connect', () => {

                    dot.className = 'status-dot online';

                    txt.innerText = '云端实时互联中';

                    mqttClient.subscribe(topic, { qos: 1 });

                    

                    // 连上后向云端发送上线同步探测请求

                    broadcastSyncAction({ type: 'PING_REQUEST' });

                mqttClient.on('message', (t, payload) => {

                    try {

                        const data = JSON.parse(payload.toString());

                        handleCloudBroadcast(data);

                    }

                });

                txt.innerText = '离线运行中';

            }

        }


        function broadcastSyncAction(actionPayload) {

            actionPayload.fromClientId = mqttClient ? mqttClient.options.clientId : 'local';

            actionPayload.timestamp = Date.now();

            

            // 本地持久化

            localStorage.setItem(`family_cache_${roomCode}`, JSON.stringify(globalState));


            if (mqttClient && mqttClient.connected) {

                const topic = `family_consensus_hub_v1/${roomCode}`;

                mqttClient.publish(topic, JSON.stringify(actionPayload), { qos: 1 });

            }

        }


        function handleCloudBroadcast(data) {

            // 忽略自己发送的回环广播

                globalState = data.state;

                localStorage.setItem(`family_cache_${roomCode}`, JSON.stringify(globalState));

                renderAllComponents();

        async function callRealAiApi(userMsg, roomRolePrompt) {

            const apiKey = localStorage.getItem('hub_api_key');

            let baseUrl = localStorage.getItem('hub_api_url') || 'https://api.deepseek.com/v1';

            const model = localStorage.getItem('hub_api_model') || 'deepseek-chat';


            if (!baseUrl.endsWith('/chat/completions')) {

                baseUrl = baseUrl.replace(/\/+$/, '') + '/chat/completions';

            }


            const profileTexts = globalState.profileList.map(item => `${item.title}: ${item.desc}`).join('\n');

            const systemPrompt = `【你在当前窗口的专属角色定位与指令】：

${roomRolePrompt}


【通用背景与家庭底牌数据】：

${profileTexts}


【输出格式要求】：

请务必使用通俗、口语化的大白话，并严格输出以下 JSON 格式：

{

  "flawAnalysis": "结合你的角色设定，客观指出刚才发言中谁在哪句话有什么逻辑问题或需要注意的地方（大白话直接指明）",

  "currentEvent": "一句话概括当前争执的核心事件",

  "actualDemands": "双方各自真正想要的实际需求（不带情绪）",

  "nextStep": "具体的一步行动建议"

}`;


            const response = await fetch(baseUrl, {

                method: 'POST',

                headers: { 'Content-Type': 'application/json', 'Authorization': `Bearer ${apiKey}` },

                body: JSON.stringify({

                    model: model,

                    messages: [{ role: 'system', content: systemPrompt }, { role: 'user', content: userMsg }],

                    temperature: 0.3

                })

            });


            if (!response.ok) throw new Error(`API 响应错误：${response.statusText}`);

            const data = await response.json();

            const rawContent = data.choices[0].message.content;

            

            try {

                const cleaned = rawContent.replace(/```json/g, '').replace(/```/g, '').trim();

                return JSON.parse(cleaned);

            } catch (e) {

                return { flawAnalysis: rawContent, currentEvent: "实时交流梳理", actualDemands: "互相听懂对方诉求", nextStep: "一次只谈一件事" };

            }

        }
